Supabase 실시간 스터디 동기화

같은 Supabase 프로젝트 + 방 코드로 연결된 스터디원끼리 카드 수정·추가·댓글이 실시간 반영됩니다. 복습 타이머(1분·5분·10분·1일)와 암기 진도는 각자 기기에 독립 저장됩니다.

처음 설정하기 (무료 Supabase 1분 세팅)
  1. supabase.com에서 무료 프로젝트 생성 → 좌측 SQL Editor에 아래 코드를 붙여넣고 Run 클릭
  2. Project Settings → API에서 Project URL과 anon / publishable key를 복사해 위에 입력
  3. 스터디원들에게 URL, Key, 방 코드만 공유하면 모든 카드 수정이 실시간 동기화됩니다.
create table if not exists public.mc_cards (
  id text primary key,
  room_code text not null,
  subject_name text not null default '미분류',
  prof_name text not null default '담당 교수',
  chapter_name text not null default '1장',
  front text not null,
  back text not null default '',
  tags jsonb not null default '[]'::jsonb,
  starred boolean not null default false,
  author text not null default '스터디원',
  updated_by text not null default '스터디원',
  comments jsonb not null default '[]'::jsonb,
  updated_at timestamptz not null default now()
);

alter table public.mc_cards enable row level security;
drop policy if exists "mc_room_all" on public.mc_cards;
create policy "mc_room_all" on public.mc_cards for all using (true) with check (true);
alter publication supabase_realtime add table public.mc_cards;
